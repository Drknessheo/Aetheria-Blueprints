import os
import subprocess
import shutil
from pathlib import Path

# Our shared neural garden
PLAYGROUND = Path.home() / "playground"

# The DNA Mapping: Auto-detecting the language
SIGNA_MAP = {
    "python": {".py": "python"},
    "nodejs": {".js": "node"},
    "c": {".c": "gcc {f} -o {n} && ./{n}"},
    "rust": {".rs": "rustc {f} -o {n} && ./{n}"}
}

class AetheriaOS:
    def __init__(self):
        PLAYGROUND.mkdir(exist_ok=True)
        self.header = "« AETHERIA SYMBIONT | PLAYGROUND »"

    def refresh_view(self):
        os.system('clear')
        print(f"\033[1;35m{self.header}\033[0m")
        print("Connected: Host | Status: Synced\n")

    def detect_and_save(self, name, code):
        # Gut check: What is this code?
        ext = ".py" # Default
        if "console.log" in code: ext = ".js"
        elif "#include <stdio.h>" in code: ext = ".c"
        elif "fn main()" in code: ext = ".rs"
        
        project_dir = PLAYGROUND / name
        project_dir.mkdir(exist_ok=True)
        file_path = project_dir / f"main{ext}"
        
        with open(file_path, "w") as f:
            f.write(code)
        return file_path, ext

    def animate(self, name):
        project_dir = PLAYGROUND / name
        if not project_dir.exists():
            print(f"[-] No memory found for project: {name}")
            return

        # Find the main file
        main_file = next(project_dir.glob("main.*"), None)
        if not main_file: return

        ext = main_file.suffix
        cmd = ""
        for lang, mapping in SIGNA_MAP.items():
            if ext in mapping:
                cmd = mapping[ext].format(f=main_file, n=main_file.stem)
        
        print(f"\033[1;32m[*] Animating {name}...\033[0m")
        subprocess.run(cmd, shell=True, cwd=project_dir)

    def list_apps(self):
        apps = [d.name for d in PLAYGROUND.iterdir() if d.is_dir()]
        print("\033[1;34m[Stored Memories]\033[0m")
        for app in apps:
            print(f" -> {app}")

    def purge(self, name):
        shutil.rmtree(PLAYGROUND / name)
        print(f"[!] {name} has been excised from the playground.")

    def shell(self):
        while True:
            self.refresh_view()
            cmd = input("Aetheria >> ").strip().split()
            if not cmd: continue
            
            action = cmd[0].lower()
            
            if action == "ingest": # Accepts paste
                name = input("Project Name: ")
                print("Paste code (Ctrl+D to finalize):")
                code = []
                while True:
                    try: line = input(); code.append(line)
                    except EOFError: break
                self.detect_and_save(name, "\n".join(code))
                
            elif action == "play":
                self.animate(cmd[1])
                input("\nPress Enter to return...")
            elif action == "list":
                self.list_apps()
                input("\nPress Enter to return...")
            elif action == "remove":
                self.purge(cmd[1])
            elif action == "exit":
                break

if __name__ == "__main__":
    os_manager = AetheriaOS()
    os_manager.shell()
